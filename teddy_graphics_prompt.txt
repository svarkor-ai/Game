GLSL vertex shader (shader.glsl) — standard mesh rendering:
```glsl
#version 430 core

layout(location = 0) in vec3 aPos;
layout(location = 1) in vec3 aColor;
layout(location = 2) in vec2 aUV;

uniform mat4 uProjection;
uniform mat4 uModel;

out vec3 vColor;
out vec2 vUV;

void main()
{
    gl_Position = uProjection * uModel * vec4(aPos, 1.0);
    vColor = aColor;
    vUV = aUV;
}
```

GLSL fragment shader (shader.frag):
```glsl
#version 430 core

in vec3 vColor;
in vec2 vUV;

out vec4 FragColor;

uniform sampler2D uTexture;
uniform int uHasTexture;

void main()
{
    if (uHasTexture == 1)
    {
        vec4 texColor = texture(uTexture, vUV);
        FragColor = texColor * vec4(vColor, 1.0);
    }
    else
    {
        FragColor = vec4(vColor, 1.0);
    }
}
```

### 4. ICamera (Djurspel.Graphics)
Isometrisk kameraprojektion: 2:1 ratio (bredd 64px, höjd 32px).
```csharp
public interface ICamera
{
    Vec3 Position { get; set; }
    float Zoom { get; set; }
    Vec2 WorldToScreen(Vec3 worldPos);
    Vec3 ScreenToWorld(Vec2 screenPos);
    Vec2 TileToScreen(Vec3I tile, float tilePixelWidth = 64f, float tilePixelHeight = 32f);
    Vec3I ScreenToTile(Vec2 screen, float tilePixelWidth = 64f, float tilePixelHeight = 32f);
    IEnumerable<Vec3I> GetDepthSortedOrder(IEnumerable<Vec3I> entities);
    void FollowEntity(Entity target, float smoothingFactor = 5f, float dt = 0.016f);
}
```
Isometrisk transform:
- `screenX = (tileX - tileY) * tilePixelWidth / 2 + offsetX`
- `screenY = (tileX + tileY) * tilePixelHeight / 2 - tileZ * tilePixelHeight + offsetY`
- Depth sort: `screenY` (lower = draw first)
